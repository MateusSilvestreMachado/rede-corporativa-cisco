! ============================================
! Switches de Acesso (Camada 2) - Prédio 2
! Switch 3 = HR (VLAN 40) | Switch 4 = SALES (VLAN 50) | Switch 5 = FINANCE (VLAN 60)
! ============================================

! --- Configuração COMUM aos Switches 3, 4 e 5 ---
! Uplinks em trunk carregando as 3 VLANs de departamento + VLAN 100 (servidores)
configure terminal
interface range fa0/1-2
 switchport mode trunk
 switchport trunk allowed vlan 40,50,60

interface range fastEthernet 0/1-2
 switchport mode trunk
 switchport trunk allowed vlan add 100


! ============================================
! Switch 3 - VLAN 40 (HR)
! ============================================
configure terminal
vlan 40
 name HR


! ============================================
! Switch 4 - VLAN 50 (SALES)
! ============================================
configure terminal
vlan 50
 name SALES


! ============================================
! Switch 5 - VLAN 60 (FINANCE)
! ============================================
configure terminal
vlan 60
 name FINANCE
