! ============================================
! Switch Multilayer 2 - Prédio 2 (Filial)
! Papel: Core L3 ativo (HSRP priority mais alta nas VLANs 40/50/60)
! ============================================

configure terminal

! --- Trunks para os switches de acesso (VLANs de departamento) ---
interface range fa0/1-3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 40,50,60

! --- Trunk adicional para a VLAN local de servidores (VLAN 100) ---
interface range fastEthernet 0/1-3, gigabitEthernet 0/2
 switchport mode trunk
 switchport trunk allowed vlan add 100

! --- VLANs locais ---
vlan 40
 name HR
vlan 50
 name SALES
vlan 60
 name FINANCE
vlan 100
 name SERVERS

! --- Interfaces VLAN (SVIs) - Departamentos ---
interface vlan 40
 ip address 192.168.40.2 255.255.255.0
 no shutdown
 standby 40 ip 192.168.40.1
 standby 40 priority 210
 standby 40 preempt

interface vlan 50
 ip address 192.168.50.2 255.255.255.0
 no shutdown
 standby 50 ip 192.168.50.1
 standby 50 priority 210
 standby 50 preempt

interface vlan 60
 ip address 192.168.60.2 255.255.255.0
 no shutdown
 standby 60 ip 192.168.60.1
 standby 60 priority 210
 standby 60 preempt

! --- SVI da VLAN de Servidores local (segmento próprio do Prédio 2) ---
interface vlan 100
 ip address 192.168.110.2 255.255.255.0
 standby 100 ip 192.168.110.1
 standby 100 priority 210
 standby 100 preempt
 no shutdown

! --- Roteamento L3 entre switches de core (link ponto a ponto) ---
interface g0/1
 no switchport
 ip address 10.10.2.2 255.255.255.252
 no shutdown

ip routing

! --- OSPF ---
router ospf 1
 network 10.10.2.0 0.0.0.3 area 0
 network 10.10.2.4 0.0.0.3 area 0
 network 192.168.40.0 0.0.0.255 area 0
 network 192.168.50.0 0.0.0.255 area 0
 network 192.168.60.0 0.0.0.255 area 0
 passive-interface vlan 40
 passive-interface vlan 50
 passive-interface vlan 60
 passive-interface vlan 100

! NOTA: a VLAN 100 aqui usa a sub-rede 192.168.110.0/24 (diferente da
! 192.168.100.0/24 usada no Prédio 1). Nos documentos originais do projeto
! não há um servidor físico documentado nesse segmento — trate-a como
! reserva local de servidores para a filial, a integrar futuramente.
